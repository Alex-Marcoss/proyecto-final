package juego;

public class Herramienta extends Item {

	int durabilidad;
	int daño;
	
	
	public Herramienta(String nombre, int durabilidad, int daño) {
		super(nombre);
		this.durabilidad = durabilidad;
		this.daño = daño;
		
	}
	
	 public boolean usar() {
	        if (durabilidad > 0) {
	        	
	            durabilidad--;
	            System.out.println(nombre + " usada.");
	            return true;
	            
	        } 
	        else {
	            System.out.println(nombre + " está rota.");
	            return false;
	        }
	    }
	 
	  // @Override
	  public void mostrarInfo() {
	        System.out.println(nombre + " | Durabilidad: " + durabilidad);
	    }
	
	 
	  public int getDurabilidad() {
	        return durabilidad;
	    }
	  
	
	
}
