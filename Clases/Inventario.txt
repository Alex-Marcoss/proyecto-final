package juego;

public class inventario{
	
	Item[] items; 
	int cantidad;
	int i;
	int cantRecurso;
	
	public inventario() {
		
		this.items = new Item[10];
		cantidad = 0;
	}
	
	public void agregarItem(Item item) {
		
		if(cantidad < items.length) {
			items[cantidad] = item;	
			cantidad++;
		}
	}
	
	public void mostrarIventario() {

	    for(i = 0; i < cantidad; i++) {
	        System.out.println(items[i].getNombre());
	        
	        if(items[i] instanceof Recursos) {

	            Recursos recurso = (Recursos) items[i];

	            System.out.println(
	                recurso.getNombre() + ": " + recurso.getCantidad()
	            );

	        } else {

	            System.out.println(items[i].getNombre());

	        }	
	    }
	}	
	
}
